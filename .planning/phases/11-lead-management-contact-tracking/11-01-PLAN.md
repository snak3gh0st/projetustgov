   ```sql
   CREATE TABLE IF NOT EXISTS contact_notes (
     id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
     lead_cnpj VARCHAR(20) NOT NULL,
     vendedor_id UUID REFERENCES users(id) ON DELETE CASCADE,
     tipo VARCHAR(50) NOT NULL CHECK (tipo IN ('ligacao', 'email', 'whatsapp', 'reuniao', 'outro')),
     observacao TEXT,
     created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
     updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   );
   CREATE INDEX IF NOT EXISTS idx_contact_notes_lead_cnpj ON contact_notes(lead_cnpj);
   CREATE INDEX IF NOT EXISTS idx_contact_notes_vendedor ON contact_notes(vendedor_id);
   ```
