1. Add LEFT JOIN to proponentes table to check registration status:
   ```sql
   SELECT
     vp.*,
     u.nome as vendedor_nome,
     ec.cnpj IS NOT NULL as is_existing_client,
     p.cnpj IS NULL as is_max_priority,
     p.total_convenios as executed_count,
     (
       SELECT COUNT(DISTINCT vp2.nr_emenda)
       FROM vendedor_projetos vp2
       WHERE vp2.cnpj = vp.cnpj
     ) as emenda_count,
     (
       SELECT SUM(vp2.valor_emenda)
       FROM vendedor_projetos vp2
       WHERE vp2.cnpj = vp.cnpj
     ) as total_valor_emendas
   FROM vendedor_projetos vp
   LEFT JOIN users u ON vp.vendedor_id = u.id
   LEFT JOIN existing_clients ec ON vp.cnpj = ec.cnpj
   LEFT JOIN proponentes p ON vp.cnpj = p.cnpj
   WHERE ${conditions.join(' AND ')}
   ORDER BY vp.cnpj, vp.valor_emenda DESC NULLS LAST
   ```

```typescript
export function formatParlamentarSummary(leads: VendedorProjeto[], cnpj: string): string {
  const cnpjLeads = leads.filter(l => l.cnpj === cnpj)
  if (cnpjLeads.length === 1) {
    return cnpjLeads[0].parlamentar || '-'
  }
  const parlamentares = [...new Set(cnpjLeads.map(l => l.parlamentar).filter(Boolean))]
  return parlamentares.length > 0
    ? `${parlamentares.length} parlamentar${parlamentares.length > 1 ? 'es' : ''}`
    : '-'
}
```
